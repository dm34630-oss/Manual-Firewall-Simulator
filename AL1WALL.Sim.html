function ipToInt(ip){

 if(!ip) return null;

 const parts = ip.trim().split('.');

 if(parts.length!==4) return null;

 let num = 0;

 for(let i=0;i<4;i++) {

   const p = parseInt(parts[i],10);

   if (Number.isNaN(p) || p<0 || p>255) return null;

   num = (num<<8)+p;

 }

 return num >>> 0;

}

function cidrMatch (ip, cidr) {

 if(!cidr) return false;

 if(cidr.indexOf('/')>-1){

   const [net,maskStr] = cidr.split('/');

   const mask = parseInt(maskStr,10);

   const ipn = ipToInt(ip);

   const netn = ipToInt(net);

   if(ipn===null || netn===null) return false;

   if (mask<0 || mask>32) return false;

   if (mask===0) return true;

   const shift = 32-mask;

   return (ipn>>>shift) === (netn>>>shift);

 } else if (cidr==='*' || cidr.toUpperCase()==='ANY') {

   return true;

 } else {

   return ip===cidr;

 }

}

let rules = [];

const rulesTableBody = document.querySelector('#rules-table tbody');

const logEl = document.getElementById('log');

const resultBox = document.getElementById('result-box');

const statAllow = document.getElementById('stat-allow');

const statDeny = document.getElementById('stat-deny');

const statTotal = document.getElementById('stat-total');

let countAllow = 0, countDeny = 0, countTotal = 0;

function renderRules() {

 rulesTableBody.innerHTML='';

 rules.forEach((r,i)=>{

   const tr = document.createElement('tr');

   tr.innerHTML = `<td>${i+1}</td>

   <td><span class="pill ${r.action==='allow'?'allow':'deny'}">${r.action.toUpperCase()}</span></td>

   <td>${r.src}</td>

   <td>${r.port}</td>

   <td>${r.proto}</td>

   <td>${r.comment||''}</td>

   <td><button data-i="${i}" class="example-btn">Del</button></td>`;

   rulesTableBody.appendChild(tr);

 });

 rulesTableBody.querySelectorAll('button[data-i]').forEach(btn=>btn.addEventListener('click',e=>{

   const i = parseInt(btn.getAttribute('data-i'),10);

   rules.splice(i,1);

   renderRules();

 }));

}

function addRuleFromForm(){

 const action = document.getElementById('action-select').value;

 const src = document.getElementById('src-input').value.trim() || 'ANY';

 const port = document.getElementById('port-input').value.trim() || '*';

 const proto = document.getElementById('proto-select').value || 'ANY';

 const comment = document.getElementById('comment-input').value.trim();

 rules.push({action,src,port,proto,comment});

 renderRules();

}

});

document.getElementById('shuffle-top').addEventListener('click', ()=>{

 const idx = rules.findIndex(r=>r.action==='deny');

 if(idx>0) {

   const r = rules.splice(idx,1)[0];

   rules.unshift(r);

   renderRules();

 }

});

function matchRule(sim) {

 for(let i=0;i<rules.length;i++){

   const r = rules[i];

   const srcMatch = cidrMatch(sim.src, r.src);

   const portMatch = (r.port==='*' || r.port==='' || r.port.toUpperCase()==='ANY') ? true : (String(sim.port)===String(r.port));

   const protoMatch = (r.proto==='ANY' || !r.proto) ? true : (r.proto.toUpperCase()===sim.proto.toUpperCase());

   if(srcMatch && portMatch && protoMatch) {

     return {ruleIndex:i, rule:r, action:r.action, matched:true, explanation: `Matched rule #${i+1}`};

   }

 }

 return {ruleIndex:null, rule:null, action:'deny', matched:false, explanation: 'No matching rule - default DENY'};

}

function appendLog(text){

 const time = new Date().toLocaleTimeString();

 logEl.innerHTML = `[${time}] ${text}\n` + logEl.innerHTML;

}

document.getElementById('simulate').addEventListener('click', () => {

 const src = document.getElementById('sim-src').value.trim();

 const port = document.getElementById('sim-port').value.trim() || '*';

 const proto = document.getElementById('sim-proto').value || 'ANY';

 const note = document.getElementById('sim-note').value || '';

 if(!src) { alert('Enter a source IP to simulate'); return; }

 const sim = {src, port, proto, note};

 const res = matchRule(sim);

 countTotal++;

 statTotal.textContent = countTotal;

 if(res.action==='allow') {

   countAllow++;

   statAllow.textContent = countAllow;

   resultBox.textContent = 'ALLOWED: ' + res.explanation;

   resultBox.className='pill allow';

   appendLog(`ALLOWED: ${src}:${port} proto=${proto} (${note}) - ${res.explanation}`);

 } else {

   countDeny++;

   statDeny.textContent = countDeny;

   resultBox.textContent = 'DENIED: ' + res.explanation;

   resultBox.className='pill deny';

   appendLog(`DENIED: ${src}:${port} proto=${proto} (${note}) - ${res.explanation}`);

 }

});

});
